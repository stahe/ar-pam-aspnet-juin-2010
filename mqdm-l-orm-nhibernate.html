    6. مقدمة إلى ORM NHibernate

    هذا الفصل هو مقدمة موجزة لـ NHibernate، وهو المكافئ لـ .NET في إطار عمل Java Hibernate. للحصول على مقدمة كاملة، يمكن قراءة:

    العنوان: NHibernate in Action، المؤلف: Pierre-Henri Kuaté، الناشر: Manning، ISBN-13: 978-1932394924

    ORM (Object Relational Mapper) هو مجموعة من المكتبات التي تسمح لبرنامج يستخدم قاعدة بيانات باستغلالها دون إصدار أوامر SQL صريحة ودون معرفة خصائص SGBD المستخدم.

    6.1. مكان NHIBERNATE في بنية .NET ذات الطبقات

    يمكن تصميم تطبيق .NET الذي يستخدم قاعدة بيانات في بنية طبقات على النحو التالي:

    يتواصل الطبقة [dao] مع SGBD عبر API ADO.NET (انظر الفقرة 3.3).في البنية السابقة، يرتبط الموصل [ADO.NET] بـ SGBD. وبالتالي، فإن الفئة التي تنفذ واجهة [IDbConnection] هي:

    • الفئة [MySQLConnection] لـ SGBD MySQL
    • الفئة [SQLConnection] لـ SGBD SQLServer

    وبالتالي، فإن الطبقة [dao] تعتمد على SGBD المستخدم. بعض الأطر (Linq، Ibatis.net، NHibernate) تزيل هذا القيد بإضافة طبقة إضافية بين الطبقة [dao] والموصل [ADO.NET] الخاص بـ SGBD المستخدم. سنستخدم هنا إطار العمل [NHibernate].

    فيما سبق، لم تعد الطبقة [dao] تتصل بالموصل [ADO.NET] بل بالإطار NHibernate الذي سيقدم لها واجهة مستقلة عن الموصل [ADO.NET] المستخدم. تسمح هذه البنية بتغيير SGBD دون تغيير الطبقة [dao]. في هذه الحالة، يجب تغيير الموصل [ADO.NET] فقط.

    6.3.1. تكوين الاتصال بقاعدة البيانات

    لنعد إلى بنية الاختبار:

    فيما سبق، يجب أن يتمكن [NHibernate] من الوصول إلى قاعدة البيانات. ولهذا الغرض، يحتاج إلى بعض المعلومات:

    • SGBD الذي يدير قاعدة البيانات (MySQL، SQLServer، Postgres، Oracle، ...). أضاف معظم SGBD إلى لغة SQL امتدادات خاصة بهم. من خلال معرفة SGBD، يمكن لـ NHibernate تكييف الأوامر SQL التي يصدرها مع هذا SGBD. يستخدم NHibernate مفهوم اللهجة SQL.
    • معلمات الاتصال بقاعدة البيانات (اسم قاعدة البيانات، اسم المستخدم المالك للاتصال، كلمة مروره)

    يمكن وضع هذه المعلومات في ملف التكوين [App.config]. فيما يلي الملف الذي سيتم استخدامه مع قاعدة بيانات MySQL 5:

    
<?xml version="1.0" encoding="utf-8" ?>
<configuration>
    <!-- أقسام التكوين -->
    <configSections>
        <section name="log4net" type="log4net.Config.Log4NetConfigurationSectionHandler,log4net" />
        <section name="hibernate-configuration" type="NHibernate.Cfg.ConfigurationSectionHandler, NHibernate" />
    </configSections>


    <!-- التكوين NHibernate -->
    <hibernate-configuration xmlns="urn:nhibernate-configuration-2.2">
        <session-factory>
            <property name="connection.provider">NHibernate.Connection.DriverConnectionProvider</property>
            <!--
            <property name="connection.driver_class">NHibernate.Driver.MySqlDataDriver</property>
            -->
            <property name="dialect">NHibernate.Dialect.MySQL5Dialect</property>
            <property name="connection.connection_string">
                Server=localhost;Database=dbpam_nhibernate;Uid=root;Pwd=;
            </property>
            <property name="show_sql">false</property>
            <mapping assembly="pam-nhibernate-demos"/>
        </session-factory>
    </hibernate-configuration>

    <!-- يحتوي هذا القسم على إعدادات تكوين log4net -->
    <!-- NOTE IMPORTANTE: السجلات غير نشطة بشكل افتراضي. يجب تفعيلها برمجياً باستخدام الأمر log4net.Config.XmlConfigurator.Configure();
    ! -->
    <log4net>
        <!-- تحديد ملحق الإخراج (المكان الذي يمكن أن تذهب إليه السجلات) -->
        <appender name="LogFileAppender" type="log4net.Appender.FileAppender, log4net">
            <param name="File" value="log.txt" />
            <param name="AppendToFile" value="false" />
            <layout type="log4net.Layout.PatternLayout, log4net">
                <param name="ConversionPattern" value="%d [%t] %-5p %c [%x] &lt;%X{auth}&gt; - %m%n" />
            </layout>
        </appender>
        <appender name="LogDebugAppender" type="log4net.Appender.DebugAppender, log4net">
            <layout type="log4net.Layout.PatternLayout, log4net">
                <param name="ConversionPattern" value="%d [%t] %-5p %c [%x] &lt;%X{auth}&gt; - %m%n"/>
            </layout>
        </appender>
        <appender name="ConsoleAppender" type="log4net.Appender.ConsoleAppender, log4net">
            <layout type="log4net.Layout.PatternLayout, log4net">
                <param name="ConversionPattern" value="%d [%t] %-5p %c [%x] &lt;%X{auth}&gt; - %m%n"/>
            </layout>
        </appender>

        <!-- إعداد الفئة الجذرية، وتعيين مستوى الأولوية الافتراضي وإضافة الملحقات (المكان الذي ستذهب إليه السجلات) -->
        <root>
            <priority value="INFO" />
            <!--
            <appender-ref ref="LogFileAppender" />
            <appender-ref ref="LogDebugAppender"/>
            -->
            <appender-ref ref="ConsoleAppender"/>
        </root>

        <!-- تحديد المستوى لبعض مساحات الأسماء المحددة -->
        <!-- يمكن أن يكون المستوى: ALL، DEBUG، INFO، WARN، ERROR، FATAL، OFF -->
        <logger name="NHibernate">
            <level value="INFO" />
        </logger>
    </log4net>
</configuration>

      • Server=localhost;: يقع SGBD على نفس الجهاز الذي يوجد عليه العميل الذي يحاول فتح الاتصال
      • Database=dbpam_nhibernate; : قاعدة بيانات MySQL المستهدفة
      • Uid=root; : المستخدم الذي يفتح الاتصال هو المستخدم root
      • Pwd=;: هذا المستخدم ليس لديه كلمة مرور (حالة خاصة في هذا المثال)

      • السطر 5: يشير السمة name إلى الحقل في الفئة [Cotisations] الذي سيستقبل المفتاح الأساسي للجدول [cotisations]. يشير السمة column إلى عمود الجدول [cotisations] الذي يستخدم كمفتاح أساسي. يُستخدم السمة unsaved-value لتعريف مفتاح أساسي لم يتم إنشاؤه بعد. تسمح هذه القيمة لـ NHibernate بمعرفة كيفية حفظ كائن [Cotisations] في الجدول [cotisations]. إذا كان لهذا الكائن حقل Id=0، فسيُجري عملية SQL INSERT، وإلا فسيُجري عملية SQL UPDATE. تعتمد قيمة unsaved-value على نوع الحقل Id في الفئة [Cotisations]. وهنا، يكون من النوع int والقيمة الافتراضية لنوع int هي 0. وبالتالي، سيكون حقل Id=0 لكائن [Cotisations] لم يتم حفظه بعد (وبالتالي بدون مفتاح أساسي). لو كان الحقل Id من النوع Object أو مشتقًا منه، لكنا كتبنا unsaved-value=null.
      • السطر 6: عندما يتعين على NHibernate حفظ كائن [Cotisations] بحقل Id=0، يجب عليه إجراء عملية INSERT على قاعدة البيانات، والتي يجب أن يحصل خلالها على قيمة للمفتاح الأساسي للسجل. تحتوي معظم عمليات SGBD على طريقة خاصة بها لتوليد هذه القيمة تلقائيًا. تُستخدم العلامة <generator> لتحديد الآلية التي يجب استخدامها لتوليد المفتاح الأساسي. تشير العلامة <generator class="native"> إلى ضرورة استخدام الآلية الافتراضية لعملية SGBD المستخدمة. لقد رأينا في الفقرة 6.2 أن المفاتيح الأساسية لجداولنا الثلاثة MySQL كانت تحتوي على السمة autoincrement. أثناء عمليات INSERT، لن يقدم NHibernate أي قيمة للعمود ID في السجل المضاف، تاركًا لـ MySQL مهمة توليد هذه القيمة.
    • السطر 8: تُستخدم العلامة <version> لتعريف عمود الجدول (وكذلك الحقل المقابل في الفئة) الذي يسمح بـ"تحديد إصدار" السجلات. في البداية، تكون قيمة الإصدار 1. ويتم زيادتها في كل عملية UPDATE. من ناحية أخرى، تتم أي عملية UPDATE أو DELETE باستخدام مرشح WHERE ID= id AND VERSION=v1. وبالتالي، لا يمكن للمستخدم تعديل أو حذف كائن ما إلا إذا كان لديه الإصدار الصحيح منه. إذا لم يكن الأمر كذلك، يتم إرسال استثناء بواسطة NHibernate.
    • السطر 9: تُستخدم العلامة <property> لتعريف تعيين عمود عادي (ليس مفتاحًا أساسيًا ولا عمود إصدار). وبالتالي، يشير السطر 9 إلى أن العمود CSGRDS في الجدول [COTISATIONS] مرتبط بالخاصية CsgRds للفئة [Cotisations].

    6.4.1. الكائن SessionFactory

    تتم جميع عمليات NHibernate داخل جلسة عمل. وتسلسل العمليات النموذجي لـ NHibernate هو كما يلي:

    • فتح جلسة عمل NHibernate
    • بدء معاملة في الجلسة
    • إجراء عمليات الاستمرارية مع الجلسة (تحميل، الحصول، البحث، CreateQuery، حفظ، SaveOrUpdate، حذف)
    • تأكيد (commit) أو إلغاء (rollback) المعاملة
    • إغلاق الجلسة NHibernate

    يتم الحصول على الجلسة من مصنع من النوع [SessionFactory]. هذا المصنع هو الذي تم تكوينه بواسطة العلامة <session-factory> في ملف التكوين [App.config]:

    
    <!-- التكوين NHibernate -->

    6.4.3. الواجهة ISession

    نقدم الآن بعض طرق واجهة ISession التي تم تنفيذها بواسطة جلسة NHibernate:

    ITransaction BeginTransaction()
    تبدأ معاملة في الجلسة
    ITransaction tx=session.BeginTransaction();
    void Clear()
    تفرغ الجلسة. تصبح الكائنات التي كانت تحتوي عليها منفصلة.
    session.Clear();
    void Close()
    يغلق الجلسة. يتم مزامنة الكائنات التي كانت تحتوي عليها مع قاعدة البيانات. تتم عملية المزامنة هذه أيضًا في نهاية المعاملة. هذه الحالة الأخيرة هي الأكثر شيوعًا.
    session.Close();
    IQuery CreateQuery(string queryString)
    ينشئ استعلام HQL (لغة استعلام Hibernate) للتنفيذ لاحقًا.
    IQuery query=session.createQuery("select e from Employe e);
    void Delete(object obj)
    يحذف كائنًا. قد يكون هذا الكائن تابعًا للجلسة (مرفق) أو غير تابع لها (غير مرفق). عند مزامنة الجلسة مع قاعدة البيانات، ستُجرى عملية SQL DELETE على هذا الكائن.
    // يتم تحميل موظف من BD
    الموظف e=session.Get<الموظف>(143);
    // يتم حذفه
    session.Delete(e);
    void Flush()
    يفرض مزامنة الجلسة مع قاعدة البيانات. لا يتغير محتوى الجلسة.
    session.Flush();
    T Get<T>(object id)
    يبحث في قاعدة البيانات عن الكائن T ذي المفتاح الأساسي id. إذا لم يكن هذا الكائن موجودًا، يجعل المؤشر null.
    // يتم تحميل موظف من BD
    Employe e=session.Get<Employe>(143);
    object Save(object obj)
    يضع الكائن obj في الجلسة. لا يحتوي هذا الكائن على مفتاح أساسي قبل Save. وبعد Save، يصبح له مفتاح أساسي. عند مزامنة الجلسة، ستُجرى عملية SQL INSERT على قاعدة البيانات.
    // ننشئ موظفًا
    موظف e=new موظف(){...};
    // يتم حفظه
    e=session.Save(e);
    SaveOrUpdate(object obj)
    يتم تنفيذ عملية Save إذا لم يكن لدى obj مفتاح أساسي أو عملية Update إذا كان لديه مفتاح بالفعل.
    void Update(object obj)
    يقوم بتحديث الكائن obj في قاعدة البيانات. ثم يتم تنفيذ عملية SQL UPDATE على قاعدة البيانات.
    // يتم تحميل موظف من BD
    الموظف e=session.Get<الموظف>(143);
    // نغير اسمه
    e.Nom=...;
    // يتم تحديثه في قاعدة البيانات
    session.Update(e);

    6.4.4. الواجهة IQuery

    تسمح الواجهة IQuery بالاستعلام عن قاعدة البيانات لاستخراج البيانات منها. وقد رأينا كيفية إنشاء مثيل لها:

    IQuery query=session.createQuery("select e from Employe e);

    معلمة الطريقة createQuery هي استعلام HQL (لغة استعلام Hibernate)، وهي لغة مشابهة للغة SQL ولكنها تستعلم عن الفئات بدلاً من الجداول. يطلب الاستعلام أعلاه قائمة بجميع الموظفين. فيما يلي بعض أمثلة استعلامات HQL:

    select e from Employe e where e.Nom like 'A%'
select e from Employe order by e.Nom asc
select e from Employe e where e.Indemnites.Indice=2

    نقدم الآن بعض طرق واجهة IQuery:

    IList<T> List<T>()
    تُرجع نتيجة الاستعلام في شكل قائمة من كائنات T
    IList<Employe> employes=session.createQuery("select e from Employe e order by e.Nom asc").List<Employe>();
    IList List()
    يعرض نتيجة الاستعلام في شكل قائمة حيث يمثل كل عنصر في القائمة سطرًا من نتائج Select في شكل مصفوفة من الكائنات.
    IList lines=session.createQuery("select e.Nom, e.Prenom, e.SS from Employee").List();
    lignes[i][j] يمثل العمود j من السطر i في نوع object. وبالتالي، فإن lignes[10][1] هو نوع object يمثل الاسم الأول لشخص ما. عادةً ما تكون عمليات التحويل بين الأنواع ضرورية لاسترداد البيانات بنوعها الدقيق.
    T UniqueResult<T>()
    يعرض الكائن الأول من نتيجة الاستعلام
    Employe e=session.createQuery("select e from Employe e where e.Nom='MARTIN'").UniqueResult<Employe>();

            String numSecu = "254104940426058";
            IQuery query = session.CreateQuery(@"select e from Employe e where e.SS=:numSecu");
            Employe employe = query.SetString("numSecu", numSecu).UniqueResult<Employe>();
            if (employe != null) {
              Console.WriteLine("Employe[" + numSecu + "]=" + employe);
            } else {
              Console.WriteLine("Employe[" + numSecu + "] non trouvé...");
            }

            numSecu = "xx";
            employe = query.SetString("numSecu", numSecu).UniqueResult<Employe>();
            if (employe != null) {
              Console.WriteLine("Employe[" + numSecu + "]=" + employe);
            } else {
              Console.WriteLine("Employe[" + numSecu + "] non trouvé...");
            }

            // تثبيت المعاملة
            transaction.Commit();
          }
        }
      } catch (Exception e) {
        Console.WriteLine("L'exception suivante s'est produite : " + e.Message);
      }
    }
